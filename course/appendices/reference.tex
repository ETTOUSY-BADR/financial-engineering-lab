\section{Matrix identities}

For conformable matrices:
\[
(AB)^\transpose=B^\transpose A^\transpose,\qquad
(AB)^{-1}=B^{-1}A^{-1},
\]
\[
\frac{\partial}{\partial x}(a^\transpose x)=a,\qquad
\frac{\partial}{\partial x}(x^\transpose Ax)=(A+A^\transpose)x.
\]
If $A=A^\transpose$, the last gradient is $2Ax$.

The Sherman--Morrison formula is
\[
(A+uv^\transpose)^{-1}
=A^{-1}-
\frac{A^{-1}uv^\transpose A^{-1}}{1+v^\transpose A^{-1}u},
\]
when denominator is nonzero. Woodbury generalizes:
\[
(A+UCV)^{-1}
=A^{-1}-A^{-1}U(C^{-1}+VA^{-1}U)^{-1}VA^{-1}.
\]
These identities accelerate factor-covariance inversions and clarify shrinkage.

For a block Gaussian covariance, the Schur complement appears in conditional
variance:
\[
\Var(Y\mid X)=
\Sigma_{YY}-\Sigma_{YX}\Sigma_{XX}^{-1}\Sigma_{XY}.
\]

\section{Calculus identities}

Chain rule:
$D(f\circ g)(x)=Df(g(x))Dg(x)$. Product Itô rule:
\[
\dd(X_tY_t)=X_t\dd Y_t+Y_t\dd X_t+\dd\langle X,Y\rangle_t.
\]
Leibniz integral differentiation, under domination:
\[
\frac{\dd}{\dd\theta}\int_{a(\theta)}^{b(\theta)}
f(x,\theta)\dd x
=f(b,\theta)b'-f(a,\theta)a'
+\int_a^b\partial_\theta f(x,\theta)\dd x.
\]

Euler's theorem: if differentiable $f(cx)=c^kf(x)$ for $c>0$, then
$x^\transpose\nabla f(x)=kf(x)$.

\section{Convex-analysis identities}

The convex conjugate is
\[
f^\ast(y)=\sup_x\{y^\transpose x-f(x)\}.
\]
Fenchel--Young inequality:
$f(x)+f^\ast(y)\geq x^\transpose y$, with equality when
$y\in\partial f(x)$. For $f(x)=\frac12x^\transpose Ax$ with $A\succ0$,
$f^\ast(y)=\frac12y^\transpose A^{-1}y$.

The subdifferential of $\abs{x}$ is $\{-1\}$ for $x<0$, $[-1,1]$ for $x=0$, and
$\{1\}$ for $x>0$. This kink permits lasso coefficients to be exactly zero.

\section{Normal identities}

If $X\sim N(\mu,\sigma^2)$,
\[
\E[\e^{tX}]=\e^{t\mu+\frac12t^2\sigma^2}.
\]
For $Z\sim N(0,1)$,
\[
\E[Z\ind_{\{Z>a\}}]=\phi(a),\qquad
\E[Z\mid Z>a]=\frac{\phi(a)}{1-\Phi(a)}.
\]
For jointly Gaussian $(X,Y)$,
\[
\E[Y\mid X=x]=
\mu_Y+\Sigma_{YX}\Sigma_{XX}^{-1}(x-\mu_X),
\]
\[
\Var(Y\mid X)=
\Sigma_{YY}-\Sigma_{YX}\Sigma_{XX}^{-1}\Sigma_{XY}.
\]

\section{Likelihood reference}

For log likelihood $\ell_n(\theta)=\sum_i\log p(Y_i;\theta)$, score is
$s_n(\theta)=\partial\ell_n/\partial\theta$ and observed information is
$-\partial^2\ell_n/\partial\theta\partial\theta^\transpose$. Under correct
regularity,
\[
\sqrt n(\widehat\theta-\theta_0)
\Rightarrow N(0,I(\theta_0)^{-1}),
\quad
I(\theta_0)=\E[s_i(\theta_0)s_i(\theta_0)^\transpose].
\]
Under misspecification, quasi-MLE targets the Kullback--Leibler projection and uses
sandwich covariance $H^{-1}JH^{-1}$.

\section{Information criteria and scoring rules}

\[
\mathrm{AIC}=-2\ell(\widehat\theta)+2k,\qquad
\mathrm{BIC}=-2\ell(\widehat\theta)+k\log n.
\]
For density forecast $p_t$, log score is $-\log p_t(y_t)$. For quantile $q$,
pinball loss is $(\alpha-\ind_{\{y\leq q\}})(y-q)$. For probability $p$ and binary
outcome $y$, Brier score is $(p-y)^2$. Proper scores encourage honest probabilistic
forecasts under their target functional.

\chapter{Selected Solutions and Proof Hints}

\section{Analysis and linear algebra}

\paragraph{Convex inverse inequality.}
For positive $X$, apply Jensen to $f(x)=x^{-1}$:
$(\E X)^{-1}\leq\E[X^{-1}]$. Multiply by $\E X>0$.

\paragraph{Projection matrix.}
$P_X^\transpose=P_X$ because $X^\transpose X$ is symmetric. Multiplying $P_X$ by
itself cancels the middle $X^\transpose X$. A symmetric idempotent matrix has
eigenvalues satisfying $\lambda^2=\lambda$, hence $0$ or $1$.

\paragraph{Pseudoinverse.}
If $A=U_r\Sigma_rV_r^\transpose$, set
$A^+=V_r\Sigma_r^{-1}U_r^\transpose$. Every solution of $Ax=b$ has
$x=A^+b+z$ with $z\in\ker(A)$, orthogonal to $A^+b$, so its norm is minimized at
$z=0$.

\section{Probability}

\paragraph{Gaussian conditional law.}
Partition the quadratic form in the joint normal density and complete the square in
$y$. The remaining center is
$\mu_Y+\Sigma_{YX}\Sigma_{XX}^{-1}(x-\mu_X)$ and the Schur complement is the
conditional covariance.

\paragraph{Stochastic exponential.}
For $f(t,w)=\exp(\sigma w-\sigma^2t/2)$,
$f_t=-\sigma^2f/2$, $f_w=\sigma f$, and $f_{ww}=\sigma^2f$. Itô drift cancels,
leaving $\dd f=\sigma f\dd W$. Positivity and expectation one, or an integrability
argument, establish the martingale.

\section{Econometrics}

\paragraph{Within transformation.}
Average $y_{it}=\alpha_i+x_{it}^\transpose\beta+u_{it}$ over $t$ and subtract:
\[
y_{it}-\bar y_i=(x_{it}-\bar x_i)^\transpose\beta+(u_{it}-\bar u_i).
\]
A time-invariant regressor subtracts to zero and is collinear with unit effects.

\paragraph{Overlapping returns.}
If $R_t^{(k)}=\sum_{j=1}^k r_{t+j}$ with iid $r$, adjacent observations share
$k-h$ innovations at lag $h<k$, so
$\Cov(R_t^{(k)},R_{t+h}^{(k)})=(k-h)\sigma^2$. It is zero at $h\geq k$.

\section{Derivatives}

\paragraph{Call convexity.}
For $K_1<K_2<K_3$ with $K_2$ a convex combination, a butterfly of calls has
nonnegative terminal payoff. Its initial price must be nonnegative, which is the
discrete convexity inequality. Take strike spacings to zero for
$\partial_{KK}C\geq0$.

\paragraph{Geometric Asian.}
For monitoring dates $t_j$, average log price is Gaussian because it is a linear
combination of Brownian values. Its mean follows from each log-price mean and its
variance from
$\Cov(W_{t_i},W_{t_j})=\min(t_i,t_j)$. Exponentiation produces a lognormal
geometric average, so the call is a Black-style lognormal expectation with adjusted
mean and variance.

\section{Portfolio and risk}

\paragraph{Tangency portfolio.}
Write excess mean as
$w^\transpose\mu_e=(\Sigma^{1/2}w)^\transpose(\Sigma^{-1/2}\mu_e)$.
Cauchy--Schwarz bounds squared Sharpe by
$\mu_e^\transpose\Sigma^{-1}\mu_e$, with equality when
$\Sigma^{1/2}w$ is proportional to $\Sigma^{-1/2}\mu_e$.

\paragraph{Normal Expected Shortfall.}
For $L=\mu+\sigma Z$ and $z_\alpha=\Phi^{-1}(\alpha)$,
\[
\E[L\mid L\geq\mu+\sigma z_\alpha]
=\mu+\sigma
\frac{\int_{z_\alpha}^\infty z\phi(z)\dd z}
{1-\alpha}.
\]
Since $\phi'(z)=-z\phi(z)$, the numerator is $\phi(z_\alpha)$.

\section{Numerical methods}

\paragraph{Control variate.}
\[
\Var(X-c(Y-\mu_Y))
=\Var(X)-2c\Cov(X,Y)+c^2\Var(Y).
\]
Differentiate in $c$ to obtain $c^\star=\Cov(X,Y)/\Var(Y)$. Substitution gives
$\Var(X)(1-\rho^2)$.

\paragraph{Bisection.}
After $n$ iterations, the containing interval length is $(b-a)/2^n$, so midpoint
error is at most $(b-a)/2^{n+1}$. Choose
$n\geq\lceil\log_2((b-a)/(2\varepsilon))\rceil$.

\section{What a complete solution contains}

A complete quantitative solution does more than reach a formula. It states the
domain, establishes existence and uniqueness when relevant, names the measure and
information set, checks units, verifies limiting cases, quantifies numerical error,
and explains the economic interpretation. An empirical solution additionally
records data provenance, estimator uncertainty, point-in-time validation, costs,
and failure modes.
